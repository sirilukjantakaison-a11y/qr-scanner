/************************************************* * ระบบใบเบิกวัสดุ QR * Google Apps Script * * ใช้ร่วมกับ GitHub Pages *************************************************/ const CONFIG = { ITEM_SHEET: "MASTER_ITEM", TRANSACTION_SHEET: "TRANSACTION" }; /************************************************* * doGet * * รองรับ 3 แบบ * * 1. เปิด Web App * 2. ค้นหา Item * 3. บันทึกใบเบิก *************************************************/ function doGet(e) { try { const params = e && e.parameter ? e.parameter : {}; const action = params.action || ""; /*********************************************** * เปิดหน้า Apps Script Web App ***********************************************/ if (!action) { return HtmlService .createHtmlOutput( "

ระบบใบเบิกวัสดุ

" + "

Google Apps Script เชื่อมต่อเรียบร้อย

" ) .setTitle("ระบบใบเบิกวัสดุ"); } /*********************************************** * ค้นหา Item ***********************************************/ if (action === "item") { const itemCode = params.itemCode || ""; const item = findItem(itemCode); if (!item) { return jsonpResponse( { success: false, message: "ไม่พบ Item Code: " + itemCode }, params.callback ); } return jsonpResponse( { success: true, item: item }, params.callback ); } /*********************************************** * บันทึกใบเบิก ***********************************************/ if (action === "save") { const rawData = params.data || ""; if (!rawData) { return jsonpResponse( { success: false, message: "ไม่พบข้อมูลสำหรับบันทึก" }, params.callback ); } const formData = JSON.parse(rawData); const result = saveRequest(formData); return jsonpResponse( result, params.callback ); } /*********************************************** * ไม่พบ action ***********************************************/ return jsonpResponse( { success: false, message: "ไม่รู้จัก action: " + action }, params.callback ); } catch (error) { return jsonpResponse( { success: false, message: error.message || String(error) }, e && e.parameter ? e.parameter.callback : null ); } } /************************************************* * JSONP Response *************************************************/ function jsonpResponse(data, callback) { const json = JSON.stringify(data); /*********************************************** * ถ้ามี callback * ส่งกลับเป็น JavaScript ***********************************************/ if (callback) { return ContentService .createTextOutput( callback + "(" + json + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } /*********************************************** * ถ้าไม่มี callback ***********************************************/ return ContentService .createTextOutput(json) .setMimeType( ContentService.MimeType.JSON ); } /************************************************* * FIND ITEM * * MASTER_ITEM * * A = No. * B = Category * C = Items Code * D = Product Name-Odoo * E = Description * F = Unit of Measure * G = Location * H = Min Stock * I = Status *************************************************/ function findItem(itemCode) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName( CONFIG.ITEM_SHEET ); if (!sheet) { throw new Error( "ไม่พบ Sheet: " + CONFIG.ITEM_SHEET ); } const searchCode = normalizeCode(itemCode); if (!searchCode) { return null; } const lastRow = sheet.getLastRow(); if (lastRow < 2) { return null; } const data = sheet .getRange( 2, 1, lastRow - 1, 9 ) .getDisplayValues(); for ( let i = 0; i < data.length; i++ ) { const code = normalizeCode( data[i][2] ); if (code === searchCode) { return { itemCode: data[i][2] || "", itemName: data[i][3] || "", description: data[i][4] || "", unit: data[i][5] || "", location: data[i][6] || "", minStock: data[i][7] || "", status: data[i][8] || "" }; } } return null; } /************************************************* * Normalize Item Code *************************************************/ function normalizeCode(value) { return String(value || "") .trim() .replace(/\s+/g, "") .toUpperCase(); } /************************************************* * สร้าง Request No. * * REQ-260827-001 *************************************************/ function createRequestNo() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName( CONFIG.TRANSACTION_SHEET ); if (!sheet) { throw new Error( "ไม่พบ Sheet: " + CONFIG.TRANSACTION_SHEET ); } const today = Utilities.formatDate( new Date(), Session.getScriptTimeZone(), "yyMMdd" ); const prefix = "REQ-" + today + "-"; const lastRow = sheet.getLastRow(); if (lastRow < 2) { return prefix + "001"; } const values = sheet .getRange( 2, 2, lastRow - 1, 1 ) .getDisplayValues(); let maxNo = 0; values.forEach( function(row) { const requestNo = String(row[0] || "") .trim(); if ( requestNo.indexOf(prefix) === 0 ) { const parts = requestNo.split("-"); const number = Number(parts[2]); if ( !isNaN(number) && number > maxNo ) { maxNo = number; } } } ); return ( prefix + String(maxNo + 1) .padStart(3, "0") ); } /************************************************* * SAVE REQUEST * * TRANSACTION * * A Date/Time * B Request No. * C Employee * D Emp. ID * E Department * F Station * G Model * H Used for * I Item Code * J Item Name * K Unit * L Qty *************************************************/ function saveRequest(formData) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName( CONFIG.TRANSACTION_SHEET ); if (!sheet) { throw new Error( "ไม่พบ Sheet: " + CONFIG.TRANSACTION_SHEET ); } if ( !formData || !Array.isArray(formData.items) || formData.items.length === 0 ) { throw new Error( "ไม่มีรายการ Item" ); } const requestNo = createRequestNo(); const now = new Date(); const rows = []; formData.items.forEach( function(item) { const qty = Number(item.qty); if (!item.itemCode) { throw new Error( "พบรายการที่ไม่มี Item Code" ); } if ( !qty || qty <= 0 ) { throw new Error( "จำนวน Item ไม่ถูกต้อง: " + item.itemCode ); } rows.push([ now, requestNo, formData.employee || "", formData.empId || "", formData.department || "", formData.station || "", formData.model || "", formData.usedFor || "", item.itemCode || "", item.itemName || "", item.unit || "", qty ]); } ); const startRow = sheet.getLastRow() + 1; sheet .getRange( startRow, 1, rows.length, 12 ) .setValues(rows); let totalQty = 0; rows.forEach( function(row) { totalQty += Number(row[11]) || 0; } ); return { success: true, requestNo: requestNo, itemCount: rows.length, totalQty: totalQty }; }
